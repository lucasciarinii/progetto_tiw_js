CheckLogin ---------------------------------------------------------------------------------------------
actor Utente
participant "Browser\n(login.js)" as JS
participant "CheckLogin\n(Servlet)" as Servlet
participant "UtenteDAO" as DAO

Utente->JS: compila form e preme "Accedi"

alt campi vuoti (validazione client)
  JS->JS: mostra errore nel form
else campi validi
  JS->Servlet: POST /api/check-login\n[username, password]

  alt parametri mancanti o non validi
    Servlet-->JS: errore
    JS->JS: mostra errore nel form
  else parametri validi
    Servlet->DAO: checkCredentials(username, password)
    DAO-->Servlet: Utente | null

    alt credenziali non valide
      Servlet-->JS: errore
      JS->JS: mostra errore nel form
    else credenziali corrette
      Servlet->Servlet: invalida sessione precedente\ncrea nuova sessione con utente
      Servlet-->JS: ok { ruolo, nome, cognome }

      alt ruolo = "FORNITORE"
        JS->JS: window.location.href = "fornitore.html"
      else ruolo = "CLIENTE"
        JS->JS: window.location.href = "cliente.html"
      end
    end
  end
end

Sku -> Sku: onSubmitCreaSku(event)
Sku -> Main: nascondiMessaggi()
Sku -> Sku: validaFormSku(formData)

alt validazione client fallita
    Sku -> Main: mostraMessaggioHome(errore, "error")

        Sku -> Main: parseJsonResponse(response)
        Main --> Sku: data
        Sku -> Home: reset form crea SKU
        Sku -> Main: mostraMessaggioHome("SKU creata con successo.", "success")
        Sku -> Prodotto: caricaSkuDisponibili()

actor Fornitore

    ProdottoBuilder -> Main: mostraMessaggioHome(errore, "error")
else bozza presente
    ProdottoBuilder -> ProdottoBuilder: validaBuilder(...) + serializzaNodoBuilder(...)

    alt validazione client fallita
        ProdottoBuilder -> Main: mostraMessaggioHome(errore, "error")
    else validazione client ok
        ProdottoBuilder -> CreaProdotto: POST api/fornitore/prodotto/crea (JSON)

        CreaProdotto -> CreaProdotto: doPost() + lettura payload builder
        CreaProdotto -> CreaProdotto: valida nodo ed eliminazioni
        CreaProdotto -> ProdottoDAO: existsByCodice(root.codice)
        ProdottoDAO --> CreaProdotto: esito

        alt errore server
            CreaProdotto --> ProdottoBuilder: errore HTTP + JSON
            ProdottoBuilder -> Main: parseJsonResponse(response)
            Main --> ProdottoBuilder: eccezione
            ProdottoBuilder -> Main: mostraMessaggioHome(errore, "error")
        else payload valido
            CreaProdotto -> CreaProdotto: apre transazione

            opt eliminazioni richieste
                CreaProdotto -> ProdottoDAO: elimina prodotti richiesti

            ProdottoDAO --> CreaProdotto: prodotto completo
            CreaProdotto --> ProdottoBuilder: 200 + JSON prodottoCreato

            ProdottoBuilder -> Main: parseJsonResponse(response)
            Main --> ProdottoBuilder: data

            ProdottoBuilder -> Home: reset form-crea-composto

                ProdottoBuilder -> ProdottoDettaglio: mostraDettaglioProdottoCreato(data)
                ProdottoDettaglio -> Home: aggiorna dettaglio-content
            else nessun prodotto restituito
                ProdottoBuilder -> Home: mostra messaggio nel pannello dettaglio
            end

            par
                ProdottoBuilder -> ProdottoApi: caricaProdottiDisponibili()
                ProdottoApi --> ProdottoBuilder: ok
            else
                ProdottoBuilder -> ProdottoApi: caricaSkuDisponibili()
                ProdottoApi --> ProdottoBuilder: ok
            end
        end
    end
end

    Home -> Dettaglio: btnRimuovi
    Dettaglio -> Api: rimuoviAssociazioneProdottoSku(prodottoId, skuId)
    Api -> Servlet: POST tipoRelazione=PRODOTTO_SKU
    Servlet -> DAO: findByIdConSKU(prodottoId)
    DAO --> Servlet: prodotto

    alt richiesta non valida
        Servlet --> Api: errore JSON
        Api --> Dettaglio: errore
        Dettaglio -> Home: mostra errore
    else ok
        Servlet -> DAO: removeSKUDaProdotto(prodottoId, skuId)
        Servlet -> DAO: findByIdConSKU(prodottoId)
        DAO --> Servlet: prodottoAggiornato
        Servlet --> Api: JSON prodottoAggiornato
        Api --> Dettaglio: risposta
        Dettaglio -> Dettaglio: rerenderDaRisposta(...)
                Dettaglio -> Dettaglio: rerenderDaRisposta(...)

        par refresh liste
            Dettaglio -> Api: caricaProdottiDisponibili()
        else
            Dettaglio -> Api: caricaSkuDisponibili()
        end

        Dettaglio -> Home: messaggio successo
    end

else "-" su sottoprodotto
    Fornitore -> Home: clicca "-"
    Home -> Dettaglio: btnRimuovi
    Dettaglio -> Api: rimuoviAssociazionePadreFiglio(figlioId, padreId)
    Api -> Servlet: POST tipoRelazione=PADRE_FIGLIO
    Servlet -> DAO: findById(figlioId)
    DAO --> Servlet: figlio

    alt richiesta non valida
        Servlet --> Api: errore JSON
        Api --> Dettaglio: errore
        Dettaglio -> Home: mostra errore
    else ok
        Servlet -> DAO: removePadre(figlioId)
        Servlet -> DAO: findByIdConDiscendenti(padreId)
        DAO --> Servlet: padreAggiornato
        Servlet --> Api: JSON padreAggiornato
        Api --> Dettaglio: risposta
        Dettaglio -> Dettaglio: rerenderDaRisposta(...)
        Dettaglio -> Api: caricaProdottiDisponibili()

    end

@startuml
actor Fornitore
participant "prodotto-builder.js" as Builder
participant "prodotto.js" as ProdottoPage

Fornitore -> Builder: clic su "+" su un prodotto semplice
Builder -> Builder: mostra menu contestuale

alt SKU esistente
    Fornitore -> Builder: seleziona "SKU esistente"
    Builder -> ProdottoPage: getState()
    ProdottoPage --> Builder: state

    Builder -> Builder: trovaNodoBuilder(state.builderState, nodeClientId)

    alt nodo non valido
        Builder --> Fornitore: nessuna azione
    else cache SKU vuota
        Builder -> ProdottoPage: mostraMessaggioHome("Non ci sono SKU disponibili da associare.", "error")
    else nodo valido
        Builder -> Fornitore: prompt "Id SKU da associare"

        alt scelta annullata
            Builder --> Fornitore: nessuna modifica
        else SKU non valida
            Builder -> ProdottoPage: mostraMessaggioHome("SKU selezionata non valida.", "error")
        else SKU già associata
            Builder -> ProdottoPage: mostraMessaggioHome("Questa SKU è già associata al prodotto semplice.", "error")
        else SKU valida
            Builder -> Builder: nodo.skuList.push(SKU selezionata)
            Builder -> Builder: renderBuilder()
            Builder -> ProdottoPage: getState()
            ProdottoPage --> Builder: state aggiornato
        end
    end

else Nuova SKU
    Fornitore -> Builder: seleziona "Nuova SKU"
    Builder -> ProdottoPage: getState()
    ProdottoPage --> Builder: state

    Builder -> Builder: trovaNodoBuilder(state.builderState, nodeClientId)

    alt nodo non valido
        Builder --> Fornitore: nessuna azione
    else nodo valido
        Builder -> Fornitore: prompt "Codice"
        Builder -> Fornitore: prompt "Nome"
        Builder -> Fornitore: prompt "Descrizione tecnica"
        Builder -> Fornitore: prompt "Prezzo"

        alt dati non validi o annullati
            Builder -> ProdottoPage: mostraMessaggioHome("Dati della nuova SKU non validi.", "error")
        else codice duplicato nel nodo
            Builder -> ProdottoPage: mostraMessaggioHome("Nel nodo è già presente una SKU con questo codice.", "error")
        else dati validi
            Builder -> Builder: nodo.skuList.push(nuova SKU)
            Builder -> Builder: renderBuilder()
            Builder -> ProdottoPage: getState()
            ProdottoPage --> Builder: state aggiornato
        end
    end
end
@enduml
