CheckLogin ---------------------------------------------------------------------------------------------
actor Utente
participant "Browser\n(login.js)" as JS
participant "CheckLogin\n(Servlet)" as Servlet
participant "UtenteDAO" as DAO

Utente->JS: compila form e preme "Accedi"

alt campi vuoti (validazione client)
  JS->JS: mostra errore nel form
else campi validi
  JS->Servlet: POST /api/check-login\n[username, password]

  alt parametri mancanti o non validi
    Servlet-->JS: errore
    JS->JS: mostra errore nel form
  else parametri validi
    Servlet->DAO: checkCredentials(username, password)
    DAO-->Servlet: Utente | null

    alt credenziali non valide
      Servlet-->JS: errore
      JS->JS: mostra errore nel form
    else credenziali corrette
      Servlet->Servlet: invalida sessione precedente\ncrea nuova sessione con utente
      Servlet-->JS: ok { ruolo, nome, cognome }

      alt ruolo = "FORNITORE"
        JS->JS: window.location.href = "fornitore.html"
      else ruolo = "CLIENTE"
        JS->JS: window.location.href = "cliente.html"
      end
    end
  end
end

Salva configurazione ---------------------------------------------------------------------------------------------
actor Utente
participant "Browser\n(sceltaSku.js)" as JS
participant "SalvaConfigurazione\n(Servlet)" as Servlet
participant "ProdottoDAO" as ProdottoDAO
participant "ConfigurazioneDAO" as ConfDAO

Utente->JS: compila nome configurazione\ne seleziona le SKU
Utente->JS: click su "Salva configurazione"

JS->Servlet: POST /api/cliente/configurazione/salva\n[nomeConf, prodottoId, sku_<id>...]

alt dati mancanti o non validi
  Servlet-->JS: errore
  JS->JS: mostra messaggio di errore\nresta in sezione-scelta-sku
else dati validi
  Servlet->ProdottoDAO: findByIdConDiscendenti(prodottoId)
  ProdottoDAO-->Servlet: prodotto con discendenti
  Servlet->Servlet: raccoglie i prodotti semplici\ne valida le SKU selezionate

  alt configurazione nuova
    Servlet->ConfDAO: createConfigurazione(nomeConf, clienteId,\nprodottoId, skuScelte)
    ConfDAO-->Servlet: id configurazione
  else modifica configurazione esistente
    Servlet->ConfDAO: findById(configurazioneId)\nupdateConfigurazione(...)
    ConfDAO-->Servlet: ok
  end

  Servlet-->JS: ok { id }
  JS->JS: apre sezione-dettaglio\ncarica il dettaglio della configurazione salvata
end

Click tasto  Salva SKU:-----------------------------------------------------------
@startuml
actor Fornitore
participant "fornitore.html" as Home
participant "sku.js" as Sku
participant "main.js " as Main
participant "CreaSKU" as CreaSKU
participant "SKUDAO" as SKUDAO
participant "prodotto.js" as Prodotto

Fornitore -> Home: click su "Salva SKU"
Home -> Sku: submit form-crea-sku
Sku -> Sku: onSubmitCreaSku(event)
Sku -> Main: nascondiMessaggi()
Sku -> Sku: validaFormSku(formData)

alt validazione client fallita
    Sku -> Main: mostraMessaggioHome(errore, "error")
else validazione client ok
    Sku -> CreaSKU: POST api/fornitore/sku/crea (FormData)
    CreaSKU -> CreaSKU: legge e valida i dati ricevuti

    alt errore nella creazione della SKU
        CreaSKU --> Sku: errore HTTP + JSON
        Sku -> Main: parseJsonResponse(response)
        Main --> Sku: eccezione
        Sku -> Main: mostraMessaggioHome(errore, "error")
    else SKU creata con successo
        CreaSKU -> SKUDAO: createSKU(...)
        SKUDAO --> CreaSKU: idGenerato
        CreaSKU -> SKUDAO: findById(idGenerato)
        SKUDAO --> CreaSKU: skuCreata
        CreaSKU --> Sku: 200 + JSON skuCreata

        Sku -> Main: parseJsonResponse(response)
        Main --> Sku: data
        Sku -> Home: reset form crea SKU
        Sku -> Main: mostraMessaggioHome("SKU creata con successo.", "success")
        Sku -> Prodotto: caricaSkuDisponibili()
        Prodotto --> Sku: ok
        Sku -> Home: aggiorna dettaglio-content con dettaglio SKU creata
    end
end
@enduml

Click tasto Salva Prodotto Semplice:-----------------------------------------------------------
@startuml
actor Fornitore
participant "fornitore.html" as Home
participant "prodotto-form.js" as ProdottoForm
participant "main.js " as Main
participant "CreaProdotto" as CreaProdotto
participant "ProdottoDAO" as ProdottoDAO
participant "prodotto-dettaglio.js" as ProdottoDettaglio
participant "prodotto-api.js" as ProdottoApi

Fornitore -> Home: click su "Salva prodotto semplice"
Home -> ProdottoForm: submit form-crea-semplice
ProdottoForm -> ProdottoForm: onSubmitProdottoSemplice(event)
ProdottoForm -> ProdottoForm: legge e valida i dati del form

alt validazione client fallita
    ProdottoForm -> Main: mostraMessaggioHome(errore, "error")
else validazione client ok
    ProdottoForm -> CreaProdotto: POST api/fornitore/prodotto/crea (FormData)
    CreaProdotto -> CreaProdotto: legge e valida i dati ricevuti
    CreaProdotto -> CreaProdotto: instrada il flusso su creaProdottoSemplice(...)

    alt errore nella creazione del prodotto semplice
        CreaProdotto --> ProdottoForm: errore HTTP + JSON
        ProdottoForm -> Main: parseJsonResponse(response)
        Main --> ProdottoForm: eccezione
        ProdottoForm -> Main: mostraMessaggioHome(errore, "error")
    else prodotto semplice creato con successo
        CreaProdotto -> ProdottoDAO: createProdottoSemplice(codice, nome)
        ProdottoDAO --> CreaProdotto: prodottoId

        loop per ogni SKU selezionata
            CreaProdotto -> ProdottoDAO: addSKUToProdotto(prodottoId, skuId)
            ProdottoDAO --> CreaProdotto: ok
        end

        CreaProdotto -> ProdottoDAO: findByIdConSKU(prodottoId)
        ProdottoDAO --> CreaProdotto: prodottoCreato
        CreaProdotto --> ProdottoForm: 200 + JSON prodottoCreato

        ProdottoForm -> Main: parseJsonResponse(response)
        Main --> ProdottoForm: data
        ProdottoForm -> Home: reset form crea prodotto semplice
        ProdottoForm -> Main: mostraMessaggioHome("Prodotto semplice creato con successo.", "success")
        ProdottoForm -> ProdottoDettaglio: mostraDettaglioProdottoCreato(data)
        ProdottoDettaglio -> Home: renderDettaglioProdottoInContainer(...)

        par
            ProdottoForm -> ProdottoApi: caricaProdottiDisponibili()
            ProdottoApi --> ProdottoForm: ok
        else
            ProdottoForm -> ProdottoApi: caricaSkuDisponibili()
            ProdottoApi --> ProdottoForm: ok
        end
    end
end
@enduml

    ProdottoForm -> Main: mostraMessaggioHome(errore, "error")

   par per ogni figlio selezionato

    ProdottoApi --> ProdottoForm: dettaglio completo
end

    alt errore durante il caricamento dei prodotti selezionati
        ProdottoForm -> Main: mostraMessaggioHome(errore, "error")
    else figli caricati correttamente
        ProdottoForm -> ProdottoBuilder: nextBuilderNodeId()
        ProdottoForm -> ProdottoBuilder: mappaProdottoEsistentePerBuilder(figliCompleti)
        ProdottoForm -> ProdottoForm: crea state.builderState

        ProdottoForm -> ProdottoBuilder: renderBuilder()
        ProdottoBuilder -> Home: aggiorna dettaglio-content con il builder
        ProdottoForm -> Main: mostraMessaggioHome("Bozza del prodotto composto creata con successo.", "success")
    end
end

Click tasto Salva Prodotto Composto (Salvataggio da bozza):-----------------------------------------------------------
@startuml
actor Fornitore
participant "fornitore.html" as Home
participant "prodotto-builder.js" as ProdottoBuilder
participant "main.js" as Main
participant "CreaProdotto" as CreaProdotto
participant "ProdottoDAO" as ProdottoDAO
participant "SKUDAO" as SKUDAO
participant "prodotto-dettaglio.js" as ProdottoDettaglio
participant "prodotto-api.js" as ProdottoApi

Fornitore -> Home: click su "SALVA PRODOTTO"
Home -> ProdottoBuilder: onSalvaBuilder()

alt nessuna bozza presente
    ProdottoBuilder -> Main: mostraMessaggioHome(errore, "error")
else bozza presente
    ProdottoBuilder -> ProdottoBuilder: validaBuilder(state.builderState, 1)

    alt validazione client fallita
        ProdottoBuilder -> Main: mostraMessaggioHome(errore, "error")
    else validazione client ok
        ProdottoBuilder -> ProdottoBuilder: serializzaNodoBuilder(state.builderState)
        ProdottoBuilder -> ProdottoBuilder: aggiunge eliminaProdotti e eliminaSku
        ProdottoBuilder -> CreaProdotto: POST api/fornitore/prodotto/crea (application/json)

        CreaProdotto -> CreaProdotto: doPost()
        CreaProdotto -> CreaProdotto: riconosce il flusso JSON del builder
        CreaProdotto -> CreaProdotto: gestisciCreazioneCompostoDaJson()
        CreaProdotto -> CreaProdotto: deserializza il payload JSON
        CreaProdotto -> CreaProdotto: validaNodoBuilder(root, 1)
        CreaProdotto -> ProdottoDAO: existsByCodice(root.codice)
        ProdottoDAO --> CreaProdotto: esito
        CreaProdotto -> CreaProdotto: validaEliminazioni(...)

        alt errore di validazione server
            CreaProdotto --> ProdottoBuilder: errore HTTP + JSON
            ProdottoBuilder -> Main: parseJsonResponse(response)
            Main --> ProdottoBuilder: eccezione
            ProdottoBuilder -> Main: mostraMessaggioHome(errore, "error")
        else payload valido
            CreaProdotto -> CreaProdotto: apre transazione

            opt ci sono eliminazioni richieste
                CreaProdotto -> CreaProdotto: applicaEliminazioni(...)
                CreaProdotto -> ProdottoDAO: elimina prodotti richiesti
                CreaProdotto -> SKUDAO: elimina SKU richieste
            end

            CreaProdotto -> CreaProdotto: persistiNodoBuilder(root, null, prodottoDAO)

            group persistenza ricorsiva dell'albero
                CreaProdotto -> ProdottoDAO: crea/aggiorna composto radice
                CreaProdotto -> ProdottoDAO: collega eventuali figli al padre
                CreaProdotto -> ProdottoDAO: crea/aggiorna nodi semplici e composti
                CreaProdotto -> SKUDAO: crea eventuali nuove SKU
                CreaProdotto -> ProdottoDAO: aggiunge/rimuove associazioni prodotto-SKU
                CreaProdotto -> ProdottoDAO: scollega eventuali figli non più presenti
            end

            CreaProdotto -> CreaProdotto: commit transazione
            CreaProdotto -> ProdottoDAO: findByIdConDiscendenti(rootId)
            ProdottoDAO --> CreaProdotto: prodottoCreato completo
            CreaProdotto --> ProdottoBuilder: 200 + JSON prodottoCreato

            ProdottoBuilder -> Main: parseJsonResponse(response)
            Main --> ProdottoBuilder: data
            ProdottoBuilder -> ProdottoBuilder: builderState = null
            ProdottoBuilder -> Home: reset form-crea-composto
            ProdottoBuilder -> Main: mostraMessaggioHome("Prodotto composto salvato con successo.", "success")

            alt risposta con prodotto creato
                ProdottoBuilder -> ProdottoDettaglio: mostraDettaglioProdottoCreato(data)
                ProdottoDettaglio -> Home: aggiorna dettaglio-content
            else nessun dato restituito
                ProdottoBuilder -> Home: mostra messaggio di salvataggio nel side-panel
            end

            par
                ProdottoBuilder -> ProdottoApi: caricaProdottiDisponibili()
                ProdottoApi --> ProdottoBuilder: ok
            else
                ProdottoBuilder -> ProdottoApi: caricaSkuDisponibili()
                ProdottoApi --> ProdottoBuilder: ok
            end
        end
    end
end
@enduml
